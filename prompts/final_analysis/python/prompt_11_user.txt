```python
        hashmap = {}
        num_nodes = len(edges) + 1

        for a, b in edges:
```

```python
class Solution:
    def countGoodNodes(self, edges: List[List[int]]) -> int:

        for n1, n2 in edges:
            adj_list[n1].append(n2)
            adj_list[n2].append(n1)
        res = 0
        def recurse(parent, curr):
            nonlocal res
            sum = 0
            good = True
            good_cnt = -1
            for adj in adj_list[curr]:
                if adj != parent:
                    adj_cnt = recurse(curr, adj)
                    sum += adj_cnt
                    if adj_cnt != good_cnt and good_cnt != -1:
                        good = False
                    else:
                        good_cnt = adj_cnt
            if good:
                res += 1
            return sum + 1
        recurse(-1, 0)
        return res
```

```python
class Solution:
    def maxFrequency(self, nums: List[int], k: int, numOperations: int) -> int:
        diff = []
        cnt = Counter()
        for i in nums:
            diff.append([i - k, 1])
            diff.append([i + k + 1, -1])
```

```python
class Solution:
    def shortestDistanceAfterQueries(self, n: int, queries: List[List[int]]) -> List[int]:
        # base chain
        g = [[i + 1] for i in range(n - 1)] + [[]]
        # shortest distances from 0; start with the chain distances
        dist = list(range(n))
        ans = []
        for u, v in queries:
            g[u].append(v)
            # If the new edge can improve dist[v], propagate the improvement forward
            if dist[u] + 1 < dist[v]:

                            dist[y] = dist[x] + 1
                            q.append(y)
```

```python
            left_count[i] = left_sorted.bisect_left(nums[i])
            left_sorted.add(nums[i])
        right_sorted = SortedList()
        for i in range(n - 1, -1, -1):
            right_count[i] = right_sorted.bisect_left(nums[i])
            right_sorted.add(nums[i])
        result = 0
        for i in range(n):
            if left_count[i] >= k and right_count[i] >= k:
                result += 1
        return result
```
# Efficient Code
```python
class Solution:
    def kBigIndices(self, nums: List[int], k: int) -> int:
        n_ = len(nums)
        arr1 = [False] * n_
        ans = 0
        left_count = 0
        leftHeap = []
        right_count = 0
        rightHeap = []
        for i,n in enumerate(nums):
            if left_count < k:
                heapq.heappush(leftHeap, -n)
                left_count += 1
            else:
                if n > -1 * (leftHeap[0]):
                    arr1[i] = True
                elif n < -1 * (leftHeap[0]):
                    heapq.heappushpop(leftHeap, -n)
        for j in range(n_-1, -1, -1):
            if right_count < k:
                heapq.heappush(rightHeap, -nums[j])
                right_count += 1
            else:
                if (nums[j] > -1 * (rightHeap[0])) and arr1[j] is True:
                    ans += 1
                elif nums[j] < -1 * (rightHeap[0]):
                    heapq.heappop(rightHeap)
                    heapq.heappush(rightHeap, -nums[j])
        return ans
```
