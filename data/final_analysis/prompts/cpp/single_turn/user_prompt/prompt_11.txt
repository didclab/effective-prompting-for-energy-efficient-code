class Solution {
public:
    int bfs(int n, vector<vector<int>>& adj) {
        queue<pair<int,int>> q;
        q.push({0,0});
        unordered_set<int> visit;

        while (!q.empty()) {
            auto [node, dist] = q.front();
            q.pop();

            visit.insert(node);

            if (node == n - 1) {
                return dist;
            }

            for (int nb : adj[node]) {
                if (!visit.count(nb)) {
                    q.push({nb, dist + 1});
                }
            }
        }
        return -1;

        for (auto& q : queries) {
            int u = q[0], v = q[1];
            adj[u].push_back(v);
            ans.push_back(bfs(n, adj));

        for (int i = 0; i < n - 1; i++) {
            g.push_back({i + 1});
        }
        g.push_back({});

        vector<int> dist(n);
        for (int i = 0; i < n; i++) {
            dist[i] = i;
        }

        vector<int> ans;

        for (auto& q : queries) {
            int u = q[0], v = q[1];
            g[u].push_back(v);

                            dist[y] = dist[x] + 1;
                            qu.push(y);
                        }
                    }
                }
            }

            ans.push_back(dist[n - 1]);
        }

class Solution {
public:
    int countOfPairs(vector<int>& nums) {
        const long long M = 1000000007;

        unordered_map<long long, long long> dp;

        for (int x = 0; x <= nums[0]; x++) {
            long long key = ((long long)x << 32) | (nums[0] - x);
            dp[key] = 1;
        }

        for (int i = 1; i < nums.size(); i++) {
            int num = nums[i];
            unordered_map<long long, long long> dp1;

            for (int j = 0; j <= num; j++) {
                long long total = 0;

                for (int k = 0; k <= nums[i - 1]; k++) {
                    if (k >= j && nums[i - 1] - k <= num - j) {
                        long long key = ((long long)k << 32) | (nums[i - 1] - k);
                        if (dp.count(key)) {
                            total += dp[key];
                        }
                    }
                }

                if (total) {
                    long long newkey = ((long long)j << 32) | (num - j);
                    dp1[newkey] = total % M;
                }
            }

            dp = move(dp1);
        }

        long long result = 0;
        for (auto& p : dp) {
            result = (result + p.second) % M;
        }

        return result;
    }

class Solution {
public:
    const long long MOD = 1000000007;

    long long modPow(long long a, long long b) {
        long long res = 1;
        a %= MOD;
        while (b) {
            if (b & 1) res = (res * a) % MOD;
            a = (a * a) % MOD;
            b >>= 1;
        }
        return res;
    }

    long long modInverse(long long x) {
        return modPow(x, MOD - 2);
    }

    long long comb(long long N, long long K) {
        if (K > N) return 0;
        long long res = 1;
        for (long long i = 1; i <= K; i++) {
            res = res * (N - K + i) % MOD;
            res = res * modInverse(i) % MOD;
        }
        return res;
    }

    int countOfPairs(vector<int>& nums) {
        int n = nums.size();
        int mn = nums[0];
        int num1 = 0, num2 = mn;

        for (int i = 1; i < n; i++) {
            int num = nums[i];

            if (num < num1) return 0;

            if (num > num1 + num2) {
                num1 = num - num2;
            }

            num2 = num - num1;

            if (num2 < mn) mn = num2;
        }

        return comb(n + mn, mn) % MOD;
    }
};

        int count = 0;
        vector<int> neighbours;

        for (int children : adj_matrix[node]) {
            count += 1;
            if (children != parent) {
                neighbours.push_back(children);
                count += dfs(children, adj_matrix, node);
            }
        }

        hashmap[node] = {count, neighbours};
        return count;
    }

    int countGoodNodes(vector<vector<int>>& edges) {
        int num_nodes = edges.size() + 1;
        vector<vector<int>> adj_matrix(num_nodes);

        for (auto& e : edges) {
            int a = e[0], b = e[1];
            adj_matrix[a].push_back(b);
            adj_matrix[b].push_back(a);
        }

        dfs(0, adj_matrix, -1);

        int total_count = 0;

        for (int node = 0; node < num_nodes; node++) {
            vector<int> child_sizes;

            for (int neighbor : adj_matrix[node]) {
                if (hashmap.count(neighbor) && hashmap[neighbor].first < hashmap[node].first) {
                    child_sizes.push_back(hashmap[neighbor].first);
                }

            if (child_sizes.size() <= 1) {

                    if (child_sizes[i] != child_sizes[0]) {
                        same = false;
                        break;
                    }
                }

    int maxFrequency(vector<int>& nums, int k, int numOperations) {
        vector<pair<int,int>> diff;
        unordered_map<int,int> cnt;

        for (int i : nums) {
            diff.push_back({i - k, 1});
            diff.push_back({i + k + 1, -1});
            cnt[i] += 1;
            diff.push_back({i, 0});
        }

        sort(diff.begin(), diff.end());

        int res = 0;
        int curr = 0;

        for (auto& [idx, change] : diff) {
            curr += change;
            res = max(res, min(cnt[idx] + numOperations, curr));
        }

        return res;
    }

        int cur = 0;
        for (int i = 0; i < k && i <= maxi; i++) {
            cur += pre[i];
        }

        for (int num = 0; num <= maxi; num++) {
            cur -= pre[num];

            if (num + k <= maxi) {
                cur += pre[num + k];
            }

            if (num > 0) {
                prev += pre[num - 1];
            }

            if (num > k + 1) {
                prev -= pre[num - k - 1];
            }

            ans = max(ans, pre[num] + min(numOperations, prev + cur));
        }

        return ans;
    }
};

class Solution {
public:
    int bfs(int n, vector<vector<int>>& adj) {
        queue<pair<int,int>> q;
        q.push({0,0});
        unordered_set<int> visit;

        while (!q.empty()) {
            auto [node, dist] = q.front();
            q.pop();

            visit.insert(node);

            if (node == n - 1) {
                return dist;
            }

            for (int nb : adj[node]) {
                if (!visit.count(nb)) {
                    q.push({nb, dist + 1});
                }
            }
        }

        return -1;
    }

    vector<int> shortestDistanceAfterQueries(int n, vector<vector<int>>& queries) {
        vector<vector<int>> adj(n);

        for (int i = 1; i < n; i++) {
            adj[i - 1].push_back(i);
        }

        vector<int> ans;

        for (auto& q : queries) {
            int u = q[0];
            int v = q[1];

            adj[u].push_back(v);
            ans.push_back(bfs(n, adj));
        }

        return ans;

#include <queue>
using namespace std;

class Solution {
public:
    vector<int> shortestDistanceAfterQueries(int n, vector<vector<int>>& queries) {
        vector<vector<int>> g(n);

        for (int i = 0; i < n - 1; i++) {
            g[i].push_back(i + 1);
        }

        vector<int> dist(n);
        for (int i = 0; i < n; i++) {
            dist[i] = i;
        }

        vector<int> ans;

        for (auto& q : queries) {
            int u = q[0];
            int v = q[1];

            g[u].push_back(v);

            if (dist[u] + 1 < dist[v]) {
                dist[v] = dist[u] + 1;
                queue<int> qu;
                qu.push(v);

                while (!qu.empty()) {
                    int x = qu.front();
                    qu.pop();

                    for (int y : g[x]) {
                        if (dist[x] + 1 < dist[y]) {
                            dist[y] = dist[x] + 1;
                            qu.push(y);
                        }
                    }
                }
            }

            ans.push_back(dist[n - 1]);
        }

        return ans;
    }
};

#include <vector>
#include <algorithm>
using namespace std;

class Solution {
public:
    vector<vector<int>> permute(int n) {
        vector<vector<int>> res;
        vector<int> s(n);

        for (int i = 0; i < n; i++) {
            s[i] = i + 1;
        }

        dfs(0, n, s, res);

        sort(res.begin(), res.end());
        return res;
    }

private:
    void dfs(int start, int n, vector<int>& s, vector<vector<int>>& res) {
        if (start == n) {
            res.push_back(s);
            return;
        }

        for (int i = start; i < n; i++) {
            if (start > 0 && (s[start - 1] % 2 == s[i] % 2)) {
                continue;
            }

            swap(s[start], s[i]);
            dfs(start + 1, n, s, res);
            swap(s[start], s[i]);
        }
    }
};

#include <vector>
#include <algorithm>
using namespace std;

class Fenwick {
public:
    int n;
    vector<int> tree;

    Fenwick(int n) : n(n), tree(n + 1, 0) {}

    void update(int i, int delta) {
        for (++i; i <= n; i += i & -i)
            tree[i] += delta;
    }

    int query(int i) {
        int s = 0;
        for (++i; i > 0; i -= i & -i)
            s += tree[i];
        return s;
    }
};

class Solution {
public:
    int kBigIndices(vector<int>& nums, int k) {
        int n = nums.size();
        if (n < 2 * k + 1) return 0;

        vector<int> left_count(n), right_count(n);

        vector<int> comp = nums;
        sort(comp.begin(), comp.end());
        comp.erase(unique(comp.begin(), comp.end()), comp.end());

        auto getIndex = [&](int v) {
            return lower_bound(comp.begin(), comp.end(), v) - comp.begin();
        };

        Fenwick leftFenwick(comp.size());

        for (int i = 0; i < n; i++) {
            int idx = getIndex(nums[i]);
            left_count[i] = leftFenwick.query(idx - 1);
            leftFenwick.update(idx, 1);
        }

        Fenwick rightFenwick(comp.size());

        for (int i = n - 1; i >= 0; i--) {
            int idx = getIndex(nums[i]);
            right_count[i] = rightFenwick.query(idx - 1);
            rightFenwick.update(idx, 1);
        }

        int result = 0;
        for (int i = 0; i < n; i++) {
            if (left_count[i] >= k && right_count[i] >= k)
                result++;
        }

        return result;
    }
};

#include <vector>
#include <queue>
using namespace std;

class Solution {
public:
    int kBigIndices(vector<int>& nums, int k) {
        int n_ = nums.size();
        vector<bool> arr1(n_, false);
        int ans = 0;

        int left_count = 0;
        priority_queue<int> leftHeap;

        int right_count = 0;
        priority_queue<int> rightHeap;

        for (int i = 0; i < n_; i++) {
            int n = nums[i];

            if (left_count < k) {
                leftHeap.push(n);
                left_count++;
            }
            else {
                if (n > leftHeap.top()) {
                    arr1[i] = true;
                }
                else if (n < leftHeap.top()) {
                    leftHeap.pop();
                    leftHeap.push(n);
                }
            }
        }

        for (int j = n_ - 1; j >= 0; j--) {
            if (right_count < k) {
                rightHeap.push(nums[j]);
                right_count++;
            }
            else {
                if (nums[j] > rightHeap.top() && arr1[j]) {
                    ans++;
                }
                else if (nums[j] < rightHeap.top()) {
                    rightHeap.pop();
                    rightHeap.push(nums[j]);
                }
            }
        }

        return ans;
    }
};
