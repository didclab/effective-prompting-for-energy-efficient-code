```cpp
#include <iostream>
#include <algorithm>
using namespace std;
#include <vector>
class Solution {
public:
    int removeDuplicates(std::vector<int>& nums) {
        auto newEnd = std::unique(nums.begin(), nums.end());
        nums.erase(newEnd, nums.end());
        return nums.size();
    }
};
```
